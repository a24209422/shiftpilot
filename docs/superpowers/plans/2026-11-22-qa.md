# 問答（text2sql）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 讓右側問答面板能用中文問組織、會議、請假與班表，後端把問題轉成 SQL、在每題一個的沙盒裡執行、用一句話回答，並附可展開的表格與 SQL。

**Architecture:** 後端新增 `backend/app/modules/qa/`：`sqlguard.py` 取出並檢查 SQL，`llm.py` 呼叫模型（本機 Ollama、OpenRouter、Google），`sandbox.py` 依提問者權限整理資料、用獨立的資料庫帳號 `qa_sandbox` 開連線建暫存表並執行 SQL，`prompt.py` 組提示詞，`service.py` 串起流程，`router.py` 掛 `/api/qa`。前端 `SidePanel.tsx` 接上 API，新增 `QaMessage.tsx` 顯示回答。

**Tech Stack:** Python 3.13、FastAPI、psycopg 3、httpx、PostgreSQL 17、pytest；React 19、TanStack Query 5、Vitest、Testing Library。

**Spec:** `docs/superpowers/specs/2026-11-22-qa-design.md`（執行前先讀完）

## Global Constraints

- 縮排 2 個空白；Python 與 TypeScript 的變數、函式一律 camelCase，資料庫欄位用 snake_case。
- 註解用正體中文，只寫「為什麼」，不重複程式碼的意思。
- 模型的 SQL **只能**在 `qa_sandbox` 帳號的連線上執行；這個帳號是 `LOGIN NOINHERIT`、不是任何角色的成員、對真實資料表沒有任何權限。
- 沙盒資料的時間範圍：今天往前 90 天到往後 90 天（含頭含尾）。查詢逾時 5 秒、最多取 200 列；回答句只給模型前 30 列。
- 問題 1～300 字；SQL 失敗時重試一次。
- 錯誤代碼：`QA_NOT_CONFIGURED`（503）、`QA_MODEL_UNAVAILABLE`（503）、`QA_UNKNOWN_MODEL`（422）、`QA_QUERY_FAILED`（422，`details` 帶 `sql`、`error`）。
- 密碼只放 `.env`（`QA_DB_PASSWORD`），不寫進程式碼；`.env` 已在 `.gitignore`。
- 不存問答紀錄、不寫稽核。
- 後端測試需要 `.env` 的 `TEST_DATABASE_URL`、`QA_DB_PASSWORD`，且資料庫叢集裡已有 `qa_sandbox` 帳號（Task 1 建立）；在 `backend/` 執行 `python -m pytest`（全套約 11 分鐘）。前端在 `frontend/` 執行 `npm test`、`npm run typecheck`。
- 每個 commit 訊息最後一行：`Co-Authored-By: Claude <實際執行的模型名稱> <noreply@anthropic.com>`。

-- 只能登入與建暫存表：不給任何資料表權限、不加入任何角色，模型的 SQL 才沒有角色可以切換

EOSQL

def loadEnvFile() -> None:
  # 已經設好的環境變數不會被 .env 蓋掉；檔案不存在（例如在容器裡）也不會出錯

def buildQaDatabaseUrl(databaseUrl: str, password: str) -> str:
  # 主機、port、資料庫都跟 app 一樣，只換成沒有任何資料表權限的沙盒帳號
  parts = urlsplit(databaseUrl)
  host = parts.netloc.rsplit("@", 1)[-1]
  return urlunsplit(parts._replace(netloc=f"{qaDbUser}:{quote(password, safe='')}@{host}"))


def loadConfig() -> Config:
  databaseUrl = os.getenv("DATABASE_URL")
  if not databaseUrl:
    # 沒有連線字串就啟動，之後每個請求才各自失敗，不如一開始就明講要去哪裡設定
    raise RuntimeError("請在 .env 設定 DATABASE_URL（參考 .env.example）")
  qaPassword = os.getenv("QA_DB_PASSWORD", "").strip()
  return Config(databaseUrl=databaseUrl, qaDatabaseUrl=buildQaDatabaseUrl(databaseUrl, qaPassword) if qaPassword else None)

```python
import re

# 真正的防線是沙盒帳號沒有權限；這裡先擋掉明顯的寫入與危險函式，錯誤訊息也能回饋給模型修正
forbiddenKeywords = re.compile(
  r"\b(insert|update|delete|merge|truncate|drop|alter|create|grant|revoke|copy|vacuum|analyze|reindex|cluster|lock"
  r"|set|reset|do|call|execute|prepare|deallocate|discard|listen|notify|unlisten|comment|security|refresh|import|load|checkpoint)\b",
  re.IGNORECASE,
)
# 底線是字元的一部分，\bset\b 擋不到 set_config，所以函式另外列
forbiddenFunctions = re.compile(
  r"\b(set_config|pg_read_file|pg_read_binary_file|pg_ls_dir|pg_stat_file|lo_import|lo_export|lo_\w+|dblink\w*"
  r"|pg_terminate_backend|pg_cancel_backend|pg_reload_conf)\s*\(",
  re.IGNORECASE,
)


class UnsafeSqlError(ValueError):
  pass


def extractSql(text: str) -> str:
  # qwen3 這類推理模型會先輸出 <think> 區塊
  text = re.sub(r"<think>.*?</think>", "", text or "", flags=re.DOTALL)
  fenced = re.search(r"```(?:sql)?\s*(.*?)```", text, flags=re.DOTALL | re.IGNORECASE)
  if fenced:
    text = fenced.group(1)
  else:
    # 小模型常不照格式，在 SQL 前面加說明文字，所以從第一個 SELECT / WITH 開始取
    start = re.search(r"\b(select|with)\b", text, flags=re.IGNORECASE)
    if start:
      text = text[start.start():]
  # 分號後面通常是模型補的解釋；只取第一句也順便擋掉多語句
  return text.split(";")[0].strip()


def validateSql(sql: str) -> None:
  if not sql:
    raise UnsafeSqlError("模型沒有產生 SQL")
  if not re.match(r"^\s*(select|with)\b", sql, re.IGNORECASE):
    raise UnsafeSqlError("只允許 SELECT 查詢")
  if ";" in sql:
    raise UnsafeSqlError("只允許單一 SQL 語句")
  if forbiddenKeywords.search(sql) or forbiddenFunctions.search(sql):
    raise UnsafeSqlError("SQL 含有不允許的關鍵字或函式")
```

import os
import time

import httpx

# 沒設 num_ctx 時，預設值依 Ollama 版本而定，prompt 超過上限會被默默截掉、不報錯
ollamaNumCtx = 8192
localTimeout = 180
cloudTimeout = 60

# 每一家雲端供應商的設定都在這張表；新增供應商只要加一筆，呼叫、重試和錯誤處理共用同一份
cloudProviders = {
  "openrouter": {
    "label": "OpenRouter",
    "url": "https://openrouter.ai/api/v1/chat/completions",
    "apiKeyEnv": "OPENROUTER_API_KEY",
    "modelsEnv": "OPENROUTER_MODELS",
    "errorHints": {401: "API key 錯誤，請確認 .env 的 OPENROUTER_API_KEY", 402: "OpenRouter 餘額不足"},
  },
  # key 填錯時 Google 回 400，原文就寫得很清楚，所以只補帳單和額度這兩種原文看不出原因的狀況
  "google": {
    "label": "Google",
    "url": "https://generativelanguage.googleapis.com/v1beta/openai/chat/completions",
    "apiKeyEnv": "GOOGLE_AI_API_KEY",
    "modelsEnv": "GOOGLE_AI_MODELS",
    "errorHints": {
      403: "請確認 AI Studio 已綁定帳單，而且 key 屬於有 credits 的專案",
      429: "額度用完或呼叫太頻繁，請到 AI Studio 查看用量",
    },
  },
}
cloudRetryStatuses = {429, 500, 502, 503, 504}
cloudRetryWaits = [2, 4]


class LlmUnavailableError(Exception):
  def __init__(self, message: str):
    super().__init__(message)
    self.message = message


def parseModelList(text: str | None) -> list[str]:
  return [name.strip() for name in (text or "").split(",") if name.strip()]


def getOllamaConfig() -> dict:
  return {
    "url": os.getenv("OLLAMA_URL", "http://localhost:11434").rstrip("/"),
    "model": os.getenv("OLLAMA_MODEL", "qwen2.5:3b"),
  }


def getCloudConfig(provider: str) -> dict:
  config = cloudProviders[provider]
  return {"apiKey": os.getenv(config["apiKeyEnv"], "").strip(), "models": parseModelList(os.getenv(config["modelsEnv"]))}


def listModels() -> list[dict]:
  defaultModel = getOllamaConfig()["model"]
  localModels = list(dict.fromkeys([defaultModel, *parseModelList(os.getenv("LOCAL_MODELS"))]))
  result = [{"id": model, "label": f"{model}（本機）", "remote": False} for model in localModels]
  seen = set(localModels)
  for provider, config in cloudProviders.items():
    cloud = getCloudConfig(provider)
    # 沒有 key 就選不到這一家的模型，免得使用者選了之後才看到 401
    if not cloud["apiKey"]:
      continue
    for model in cloud["models"]:
      if model not in seen:
        seen.add(model)
        result.append({"id": model, "label": f"{model}（{config['label']}）", "remote": True})
  return result


def getProvider(model: str) -> str:
  # 只看清單、不看名稱有沒有斜線：Ollama 的模型名稱也可能是 hf.co/... 這種格式
  for provider in cloudProviders:
    cloud = getCloudConfig(provider)
    if cloud["apiKey"] and model in cloud["models"]:
      return provider
  return "ollama"


def describeError(error: Exception, model: str, provider: str) -> str:
  if provider == "ollama":
    return f"無法呼叫 Ollama：{error}。請確認 Ollama 已啟動，且已下載 {model}。"
  config = cloudProviders[provider]
  response = getattr(error, "response", None)
  if response is None:
    return f"無法呼叫 {config['label']}：{error}"
  # 閘道錯誤常回整頁 HTML，只留開頭，畫面才不會被塞滿
  detail = config["errorHints"].get(response.status_code, response.text[:300])
  return f"無法呼叫 {config['label']}（HTTP {response.status_code}）：{detail}"


def callOllama(client: httpx.Client, model: str, messages: list[dict]) -> str:
  response = client.post(
    f"{getOllamaConfig()['url']}/api/chat",
    json={
      "model": model, "messages": messages, "stream": False,
      # 溫度設 0：同一個問題盡量產生同一句 SQL，結果比較穩定也比較好除錯
      "options": {"temperature": 0, "num_ctx": ollamaNumCtx},
    },
    timeout=localTimeout,
  )
  response.raise_for_status()
  return response.json()["message"]["content"]


def callCloud(client: httpx.Client, provider: str, model: str, messages: list[dict], sleep) -> str:
  config = cloudProviders[provider]
  # 429 和 5xx 多半是暫時的，等一下再試；401、402 這類重試也沒用，直接往下報錯
  for waitSeconds in [*cloudRetryWaits, None]:
    response = client.post(
      config["url"],
      headers={"Authorization": f"Bearer {getCloudConfig(provider)['apiKey']}"},
      json={"model": model, "messages": messages, "temperature": 0},
      timeout=cloudTimeout,
    )
    if response.status_code not in cloudRetryStatuses or waitSeconds is None:
      break
    sleep(waitSeconds)
  response.raise_for_status()
  payload = response.json()
  choices = payload.get("choices") if isinstance(payload, dict) else None
  if not choices:
    raise LlmUnavailableError(f"{config['label']} 的回應沒有 choices")
  # 推理模型只回傳思考內容、或被安全機制擋下時 content 是 null，當成沒產生 SQL 交給後面處理
  return (choices[0].get("message") or {}).get("content") or ""


def callModel(model: str, messages: list[dict], client: httpx.Client | None = None, sleep=time.sleep) -> str:
  provider = getProvider(model)
  ownClient = client is None
  client = client or httpx.Client()
  try:
    if provider == "ollama":
      return callOllama(client, model, messages)
    return callCloud(client, provider, model, messages, sleep)
  except httpx.HTTPError as error:
    raise LlmUnavailableError(describeError(error, model, provider))
  finally:
    if ownClient:
      client.close()
```

  windows = shiftService.workWindows(conn, sorted({leave["employee_id"] for leave in leaves}), startAt, endAt)
  result = []
  for leave in leaves:
    label = "請假" if leave["status"] == "approved" else "請假（待核准）"
    # 跟時間軸一樣依班裁成片段，「誰有空」才不會把下班時間也算成請假
    for segmentStart, segmentEnd in shiftRules.clipWindows(windows[leave["employee_id"]], leave["start_at"], leave["end_at"]):
      result.append((leave["employee_id"], segmentStart, segmentEnd, "leave", label))
  return result


def collectSandboxData(conn: Connection, actor: Actor, today: date) -> SandboxData:
  windowStart, windowEnd = dataWindow(today)
  startAt = datetime.combine(windowStart, time.min)
  endAt = datetime.combine(windowEnd + timedelta(days=1), time.min)
  rows: dict[str, list[tuple]] = {}
  rows["departments"] = tuples(conn, "SELECT id, name, parent_id FROM departments ORDER BY id")
  rows["employees"] = tuples(
    conn, "SELECT id, employee_no, name, department_id, job_title, manager_id, hired_on FROM employees WHERE is_active ORDER BY employee_no"
  )
  rows["resources"] = tuples(conn, "SELECT id, name, kind, location, capacity FROM resources WHERE is_active ORDER BY id")
  bookings = conn.execute(
    "SELECT id, title, category, start_at, end_at FROM bookings WHERE status = 'confirmed' AND start_at < %s AND end_at > %s ORDER BY start_at, id",
    (endAt, startAt),
  ).fetchall()
  bookingIds = [booking["id"] for booking in bookings]
  rows["bookings"] = [tuple(booking.values()) for booking in bookings]
  rows["booking_resources"] = tuples(
    conn, "SELECT booking_id, resource_id FROM booking_resources WHERE booking_id = ANY(%s) ORDER BY booking_id, resource_id", (bookingIds,)
  )
  participants = conn.execute(
    "SELECT booking_id, employee_id, role FROM booking_participants WHERE booking_id = ANY(%s) ORDER BY booking_id, employee_id",
    (bookingIds,),
  ).fetchall()
  rows["booking_participants"] = [tuple(row.values()) for row in participants]
  rows["leave_requests"] = visibleLeave(conn, actor, startAt, endAt)
  rows["day_shifts"] = [
    (employee["id"], day.day, day.shift.name if day.shift else None, day.startAt, day.endAt, day.breakStartAt, day.breakEndAt, day.source)
    for employee, days in shiftService.resolveRange(conn, None, windowStart, windowEnd)
    for day in days
  ]
  bookingById = {booking["id"]: booking for booking in bookings}
  meetingBusy = [
    (row["employee_id"], bookingById[row["booking_id"]]["start_at"], bookingById[row["booking_id"]]["end_at"], "meeting", bookingById[row["booking_id"]]["title"])
    for row in participants
  ]
  rows["busy_periods"] = meetingBusy + leaveBusy(conn, startAt, endAt)
  rows["calendar_days"] = tuples(conn, "SELECT day, kind, name FROM calendar_days WHERE day BETWEEN %s AND %s ORDER BY day", (windowStart, windowEnd))
  vocabulary = {
    "departments": [row[1] for row in rows["departments"]],
    "shifts": names(conn, "SELECT name FROM shift_types WHERE is_active ORDER BY id"),
    "leaveTypes": names(conn, "SELECT name FROM leave_types WHERE is_active ORDER BY id"),
    "resources": [row[1] for row in rows["resources"]],
  }
  return SandboxData(windowStart, windowEnd, rows, vocabulary)


@contextmanager
def openSandbox(qaDatabaseUrl: str, data: SandboxData):
  # autocommit：每一句自成交易；切成唯讀之後，模型的 SQL 也在自己的唯讀交易裡執行
  with psycopg.connect(qaDatabaseUrl, autocommit=True) as conn:
    for name, columns in sandboxTables:
      conn.execute(f"CREATE TEMP TABLE {name} ({columns})")
      tableRows = data.rows[name]
      if tableRows:
        with conn.cursor().copy(f"COPY {name} FROM STDIN") as copy:
          for row in tableRows:
            copy.write_row(row)
      # 唯讀交易仍允許寫暫存表；擁有者收回自己的寫入權限後，唯讀狀態下也無法再 GRANT 回來
      conn.execute(f"REVOKE INSERT, UPDATE, DELETE, TRUNCATE ON {name} FROM CURRENT_USER")
    # 只看得到自己的暫存表；真實資料表本來就沒有權限，這裡再讓沒寫 schema 的名稱只找暫存表
    conn.execute("SET search_path = pg_temp")
    conn.execute(f"SET statement_timeout = '{statementTimeout}'")
    conn.execute("SET default_transaction_read_only = on")
    yield conn


def jsonValue(value):
  # datetime 是 date 的子類別，要先判斷
  if isinstance(value, datetime):
    return value.isoformat(timespec="seconds")
  if isinstance(value, (date, time)):
    return value.isoformat()
  if isinstance(value, Decimal):
    return float(value)
  if isinstance(value, timedelta):
    return str(value)
  return value


def runQuery(conn: Connection, sql: str) -> tuple[list[str], list[list], bool]:
  with conn.cursor() as cursor:
    cursor.execute(sql)
    if cursor.description is None:
      return [], [], False
    columns = [column.name for column in cursor.description]
    fetched = cursor.fetchmany(maxRows + 1)
  return columns, [[jsonValue(value) for value in row] for row in fetched[:maxRows]], len(fetched) > maxRows

def buildDateReference(now: datetime) -> str:
  # 小模型很不會算「下週三」是哪天，直接把日期表給它查
  today = now.date()
  monday = today - timedelta(days=today.weekday())
  lines = [
    f"今天是 {today} 星期{weekdayNames[today.weekday()]}，現在時間 {now.strftime('%H:%M')}。一週從星期一開始。",
    f"本週：{monday} ～ {monday + timedelta(days=6)}",
    f"下週：{monday + timedelta(days=7)} ～ {monday + timedelta(days=13)}",
    f"上週：{monday - timedelta(days=7)} ～ {monday - timedelta(days=1)}",
  ]
  relativeNames = {-1: "昨天", 0: "今天", 1: "明天", 2: "後天"}
  for offset in range(-1, 14):
    day = today + timedelta(days=offset)
    label = f"（{relativeNames[offset]}）" if offset in relativeNames else ""
    lines.append(f"{day} 星期{weekdayNames[day.weekday()]}{label}")
  return "\n".join(lines)

def describeSqlError(error: Exception) -> str:
  if isinstance(error, psycopg.errors.QueryCanceled):
    return "查詢太久（超過 5 秒）"
  # PostgreSQL 的錯誤第二行之後是 LINE／HINT 這類細節，給模型看第一行就夠它修正
  return str(error).strip().split("\n")[0]


def composeAnswer(llm, model: str, question: str, columns: list[str], rows: list[list], truncated: bool) -> str:
  fallback = f"查到 {len(rows)}{' 筆以上' if truncated else ' 筆'}資料。"
  try:
    reply = llm(model, prompt.buildAnswerMessages(question, columns, rows, len(rows), truncated))
  except llmModule.LlmUnavailableError:
    # 回答句只是讓結果好讀；查詢已經成功，表格照樣給
    return fallback
  answer = re.sub(r"<think>.*?</think>", "", reply or "", flags=re.DOTALL).strip()
  return answer or fallback


def ask(
  conn: Connection, actor: Actor, question: str, model: str, qaDatabaseUrl: str | None, now: datetime, llm=llmModule.callModel
) -> dict:
  if not qaDatabaseUrl:
    raise AppError("QA_NOT_CONFIGURED", "問答還沒設定：請在 .env 設定 QA_DB_PASSWORD 並建立 qa_sandbox 帳號（見 README 的「問答」）", 503)
  question = (question or "").strip()
  if not question or len(question) > maxQuestionLength:
    raise AppError("VALIDATION", f"問題要 1 到 {maxQuestionLength} 個字", 422, {"field": "question"})
  models = {item["id"]: item for item in llmModule.listModels()}
  if model not in models:
    raise AppError("QA_UNKNOWN_MODEL", "沒有這個模型", 422, {"field": "model"})
  data = sandbox.collectSandboxData(conn, actor, now.date())
  messages = prompt.buildSqlMessages(question, data, now)
  sql, lastError, result = "", "", None
  with sandbox.openSandbox(qaDatabaseUrl, data) as sandboxConn:
    for _ in range(2):
      reply = callLlm(llm, model, messages)
      sql = sqlguard.extractSql(reply)
      try:
        sqlguard.validateSql(sql)
        result = sandbox.runQuery(sandboxConn, sql)
        break
      except (sqlguard.UnsafeSqlError, psycopg.Error) as error:
        lastError = describeSqlError(error)
        # 把錯誤訊息回饋給模型重試：小模型常拼錯欄位名稱，給它看錯誤通常一次就能修好
        messages = messages + [
          {"role": "assistant", "content": reply},
          {"role": "user", "content": f"這句 SQL 失敗了：{lastError}\n請修正後只輸出一句 SQL。"},
        ]
  if result is None:
    raise AppError("QA_QUERY_FAILED", f"查詢失敗：{lastError}", 422, {"sql": sql, "error": lastError})
  columns, rows, truncated = result
  return {
    "answer": composeAnswer(llm, model, question, columns, rows, truncated),
    "sql": sql, "columns": columns, "rows": rows, "truncated": truncated,
    "model": model, "remote": models[model]["remote"],
    "window": {"from": data.windowStart.isoformat(), "to": data.windowEnd.isoformat()},
  }
```

import type { QaAnswer } from "../api/types";

export type ChatEntry =
  | { id: number; kind: "question"; text: string }
  | { id: number; kind: "pending" }
  | { id: number; kind: "answer"; result: QaAnswer }
  | { id: number; kind: "error"; message: string; sql: string | null };

const cellText = (value: unknown) => (value === null || value === undefined ? "—" : String(value));

function ResultTable({ result }: { result: QaAnswer }) {
  if (result.columns.length === 0) return <div className="empty">沒有資料</div>;
  return (
    <div className="qaTable">
      <table>
        <thead><tr>{result.columns.map((column) => <th key={column}>{column}</th>)}</tr></thead>
        <tbody>
          {result.rows.map((row, rowIndex) => (
            <tr key={rowIndex}>{row.map((value, columnIndex) => <td key={columnIndex}>{cellText(value)}</td>)}</tr>
          ))}
        </tbody>
      </table>
      {result.rows.length === 0 && <div className="empty">查無資料</div>}
      {result.truncated && <div className="chatNote">只顯示前 {result.rows.length} 筆</div>}
    </div>
  );
}

export function ChatMessage({ entry }: { entry: ChatEntry }) {
  if (entry.kind === "question") {
    return <div className="msg user"><div className="bubble">{entry.text}</div></div>;
  }
  if (entry.kind === "pending") {
    return <div className="msg bot"><div className="msgMeta">DESK</div><div className="bubble">查詢中…</div></div>;
  }
  if (entry.kind === "error") {
    return (
      <div className="msg bot">
        <div className="msgMeta">DESK</div>
        <div className="bubble qaError" role="alert">
          {entry.message}
          {entry.sql && <details><summary>看 SQL</summary><pre className="mono qaSql">{entry.sql}</pre></details>}
        </div>
      </div>
    );
  }
  const { result } = entry;
  return (
    <div className="msg bot">
      {/* 雲端模型要標出來：問題與查詢結果已經離開這台電腦 */}
      <div className="msgMeta">{result.model}{result.remote ? "・已送到雲端" : ""}</div>
      <div className="bubble">
        {result.answer}
        <details>
          <summary>看查詢結果</summary>
          <ResultTable result={result} />
          <pre className="mono qaSql">{result.sql}</pre>
        </details>
      </div>
    </div>
  );
}
