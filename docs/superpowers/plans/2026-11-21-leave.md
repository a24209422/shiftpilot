# 請假模組 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增請假模組：假別、請假申請、直屬主管核准、特休依年資計算（週年制）、年度上限、人資額度調整與未休報表；請假會出現在排會議的參加者警告與時間軸的員工檢視。

**Architecture:** 後端新增 `backend/app/modules/leave/`：`rules.py` 放純函式（年資、時數、年度），`service.py` 放假別、額度、餘額、報表，`requests.py` 放申請的流程、行事曆與「不可用」提供者，`router.py` 掛在 `/api/leave`。餘額不存，每次由規則額度＋調整－申請加總算出。前端新增 `frontend/src/modules/leave/`（請假頁、假別與額度頁），首頁加「待審核請假」方塊；時間軸加 `block` 種類、檢視選單與員工檢視。

**Tech Stack:** Python 3.13、FastAPI、psycopg 3、PostgreSQL 17（`btree_gist`）、pytest；React 19、TanStack Query 5、react-router 8、Vitest、Testing Library。

**Spec:** `docs/superpowers/specs/2026-11-21-leave-design.md`（執行前先讀完）

## Global Constraints

- 縮排 2 個空白；Python 與 TypeScript 的變數、函式一律 camelCase（Python 後端也是），資料庫欄位用 snake_case。
- 註解用正體中文，只寫「為什麼」，不重複程式碼的意思。
- 時間一律是不含時區的本地時間：資料表用 `timestamp`，API 用 `YYYY-MM-DDTHH:MM[:SS]`，用 `parseLocalTime` 解析。
- 每個寫入動作都要 `writeAudit(conn, actor, "leave", <動作>, <entity>, <id>, before, after)`。
- 錯誤用 `AppError(code, message, status, details)`，前端由統一格式顯示。
- 上班時間固定：週一到週五 09:00–12:00、13:00–18:00，一天 8 小時；`calendar_days` 的 `holiday` 不上班、`makeup_workday` 要上班。
- 請假的最小單位是 1 小時，開始、結束都要整點。
- 不寫示範用的請假資料；畫面上不能有假資料或沒有後端的按鈕。
- 密碼、金鑰只放 `.env`。
- 後端測試需要 `.env` 的 `TEST_DATABASE_URL`，並先 `docker compose up -d`；在 `backend/` 執行 `python -m pytest`。前端在 `frontend/` 執行 `npm test`、`npm run typecheck`。

CREATE TABLE leave_types (
  id int GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  code text NOT NULL UNIQUE CHECK (code ~ '^[a-z][a-z0-9_]*$'),
  name text NOT NULL UNIQUE CHECK (btrim(name) <> ''),
  -- 給之後的薪資模組參考，請假模組本身不使用
  paid_ratio numeric(3, 2) NOT NULL CHECK (paid_ratio BETWEEN 0 AND 1),
  quota_kind text NOT NULL CHECK (quota_kind IN ('seniority', 'yearly_cap', 'none')),
  annual_cap_hours int NULL CHECK (annual_cap_hours > 0),
  is_active boolean NOT NULL DEFAULT true,
  CHECK ((quota_kind = 'yearly_cap') = (annual_cap_hours IS NOT NULL))
);

-- 特休的額度由年資規則算，只能有一種
CREATE UNIQUE INDEX leave_types_one_seniority_idx ON leave_types (quota_kind) WHERE quota_kind = 'seniority';

CREATE TABLE leave_requests (
  id int GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  employee_id int NOT NULL REFERENCES employees (id),
  leave_type_id int NOT NULL REFERENCES leave_types (id),
  start_at timestamp NOT NULL,
  end_at timestamp NOT NULL,
  -- 送出時算好的快照：之後人資改了假日，已核准的時數不會跟著悄悄改變
  hours int NOT NULL CHECK (hours > 0),
  reason text NOT NULL DEFAULT '',
  status text NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'approved', 'rejected', 'withdrawn', 'cancelled')),
  decided_by int NULL REFERENCES users (id),
  decided_at timestamp NULL,
  decision_note text NULL,
  cancelled_by int NULL REFERENCES users (id),
  cancelled_at timestamp NULL,
  created_by int NOT NULL REFERENCES users (id),
  created_at timestamp NOT NULL,
  CHECK (end_at > start_at),
  -- cancelled 只會從 approved 來，所以也有核准紀錄
  CHECK ((status IN ('approved', 'rejected', 'cancelled')) = (decided_at IS NOT NULL)),
  CHECK ((decided_at IS NULL) = (decided_by IS NULL)),
  CHECK ((status IN ('withdrawn', 'cancelled')) = (cancelled_at IS NOT NULL)),
  CHECK ((cancelled_at IS NULL) = (cancelled_by IS NULL)),
  -- 同一個人還有效的申請不能重疊；由資料庫保證，兩張同時送出也一樣
  EXCLUDE USING gist (employee_id WITH =, tsrange(start_at, end_at) WITH &&) WHERE (status IN ('pending', 'approved'))
);

CREATE INDEX leave_requests_employee_start_idx ON leave_requests (employee_id, start_at);
CREATE INDEX leave_requests_status_idx ON leave_requests (status);

-- 只能新增：要更正就再加一筆反向的，稽核才看得出來改過什麼
CREATE TABLE leave_adjustments (
  id int GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  employee_id int NOT NULL REFERENCES employees (id),
  leave_type_id int NOT NULL REFERENCES leave_types (id),
  period_start date NOT NULL,
  hours int NOT NULL CHECK (hours <> 0),
  reason text NOT NULL CHECK (btrim(reason) <> ''),
  created_by int NOT NULL REFERENCES users (id),
  created_at timestamp NOT NULL
);

CREATE INDEX leave_adjustments_lookup_idx ON leave_adjustments (employee_id, leave_type_id, period_start);

```python
import calendar
from dataclasses import dataclass
from datetime import date, datetime, time, timedelta

from app.infra.errors import AppError

hoursPerDay = 8
# 排班模組做好之前，所有人都用固定的上班時間；之後改成依各自的班表計算
workWindows = ((time(9), time(12)), (time(13), time(18)))


@dataclass(frozen=True)
class Period:
  start: date
  # 含這一天
  end: date
  # None 代表不限額度
  quotaHours: int | None


def addMonths(day: date, months: int) -> date:
  yearOffset, monthIndex = divmod(day.month - 1 + months, 12)
  year = day.year + yearOffset
  month = monthIndex + 1
  # 8/31 加 6 個月沒有 2/31，依設計書的約定取該月最後一天
  return date(year, month, min(day.day, calendar.monthrange(year, month)[1]))


def annualLeaveDays(years: int) -> int:
  # 勞基法第 38 條；years 是滿幾年，至少 1
  if years == 1:
    return 7
  if years == 2:
    return 10
  if years <= 4:
    return 14
  if years <= 9:
    return 15
  return min(30, 15 + years - 9)


def seniorityPeriod(hiredOn: date, day: date) -> Period | None:
  halfYear = addMonths(hiredOn, 6)
  if day < halfYear:
    return None
  firstYear = addMonths(hiredOn, 12)
  if day < firstYear:
    return Period(halfYear, firstYear - timedelta(days=1), 3 * hoursPerDay)
  years = day.year - hiredOn.year
  if addMonths(hiredOn, 12 * years) > day:
    years -= 1
  start = addMonths(hiredOn, 12 * years)
  end = addMonths(hiredOn, 12 * (years + 1)) - timedelta(days=1)
  return Period(start, end, annualLeaveDays(years) * hoursPerDay)


def periodFor(quotaKind: str, annualCapHours: int | None, hiredOn: date, day: date) -> Period | None:
  if quotaKind == "seniority":
    return seniorityPeriod(hiredOn, day)
  # 特休以外的假別，年度上限以曆年計
  quota = annualCapHours if quotaKind == "yearly_cap" else None
  return Period(date(day.year, 1, 1), date(day.year, 12, 31), quota)


def isWorkday(day: date, calendarKinds: dict[date, str]) -> bool:
  kind = calendarKinds.get(day)
  if kind == "holiday":
    return False
  if kind == "makeup_workday":
    return True
  return day.weekday() < 5


def requireWholeHour(value: datetime, field: str) -> None:
  if value.minute or value.second or value.microsecond:
    raise AppError("VALIDATION", "請假時間必須是整點", 422, {"field": field})


def countLeaveHours(startAt: datetime, endAt: datetime, calendarKinds: dict[date, str]) -> int:
  requireWholeHour(startAt, "startAt")
  requireWholeHour(endAt, "endAt")
  total = timedelta()
  day = startAt.date()
  while day <= endAt.date():
    if isWorkday(day, calendarKinds):
      for windowStart, windowEnd in workWindows:
        start = max(startAt, datetime.combine(day, windowStart))
        end = min(endAt, datetime.combine(day, windowEnd))
        if end > start:
          total += end - start
    day += timedelta(days=1)
  return int(total.total_seconds() // 3600)


def periodForRequest(quotaKind: str, annualCapHours: int | None, hiredOn: date, startAt: datetime, endAt: datetime) -> Period:
  first = periodFor(quotaKind, annualCapHours, hiredOn, startAt.date())
  if first is None:
    raise AppError("LEAVE_NOT_ELIGIBLE", "到職未滿 6 個月，還沒有特休", 422)
  # 結束在 00:00 時，實際的最後一天是前一天
  last = periodFor(quotaKind, annualCapHours, hiredOn, (endAt - timedelta(microseconds=1)).date())
  if last != first:
    raise AppError(
      "LEAVE_CROSSES_PERIOD", f"這段時間跨了兩個年度（這個年度到 {first.end.isoformat()} 為止），請拆成兩張申請", 422,
    )
  return first
```

      (code, name, data.get("paidRatio"), data["quotaKind"], data.get("annualCapHours")),
    ).fetchone()
  leaveType = camelizeRow(row)
  writeAudit(conn, actor, "leave", "createLeaveType", "leaveType", leaveType["id"], None, leaveType)
  return leaveType


def updateLeaveType(conn: Connection, actor: Actor, leaveTypeId: int, fields: dict) -> dict:
  for frozen in ("quotaKind", "code"):
    # 額度類型一改，舊申請的年度算法就跟著變；代碼是給程式辨認的
    if frozen in fields:
      raise AppError("VALIDATION", "代碼與額度類型建立後不能修改", 422, {"field": frozen})
  unknown = set(fields) - set(typeUpdateFields)
  if unknown:
    raise AppError("VALIDATION", f"不能修改的欄位：{', '.join(sorted(unknown))}", 422)
  if "name" in fields:
    fields = {**fields, "name": requireText(fields["name"], "名稱")}
  before = getLeaveType(conn, leaveTypeId)
  if not fields:
    return before
  assignments = ", ".join(f"{typeUpdateFields[key]} = %s" for key in fields)
  with mapDbErrors(conn):
    conn.execute(f"UPDATE leave_types SET {assignments} WHERE id = %s", [*fields.values(), leaveTypeId])
  after = getLeaveType(conn, leaveTypeId)
  writeAudit(conn, actor, "leave", "updateLeaveType", "leaveType", leaveTypeId, before, after)
  return after


# ---------- 餘額 ----------


def balanceFor(conn: Connection, employeeId: int, hiredOn: date, leaveType: dict, day: date) -> dict:
  result = {
    "leaveTypeId": leaveType["id"], "leaveTypeName": leaveType["name"], "quotaKind": leaveType["quotaKind"],
    "periodStart": None, "periodEnd": None, "quotaHours": 0, "adjustmentHours": 0,
    "approvedHours": 0, "pendingHours": 0, "remainingHours": 0,
  }
  period = rules.periodFor(leaveType["quotaKind"], leaveType["annualCapHours"], hiredOn, day)
  if period is None:
    return result
  # 申請不會跨年度，所以用開始時間判斷屬於哪個年度就夠了
  used = {
    row["status"]: row["hours"]
    for row in conn.execute(
      "SELECT status, sum(hours)::int AS hours FROM leave_requests WHERE employee_id = %s AND leave_type_id = %s "
      "AND status IN ('pending', 'approved') AND start_at >= %s AND start_at < %s GROUP BY status",
      (employeeId, leaveType["id"], period.start, period.end + timedelta(days=1)),
    ).fetchall()
  }
  adjustment = conn.execute(
    "SELECT coalesce(sum(hours), 0)::int AS hours FROM leave_adjustments WHERE employee_id = %s AND leave_type_id = %s AND period_start = %s",
    (employeeId, leaveType["id"], period.start),
  ).fetchone()["hours"]
  approved = used.get("approved", 0)
  pending = used.get("pending", 0)
  # 待核准的也先扣：不然同時送好幾張，每張單獨看都沒超過上限
  remaining = None if period.quotaHours is None else period.quotaHours + adjustment - approved - pending
  result.update(
    periodStart=period.start.isoformat(), periodEnd=period.end.isoformat(), quotaHours=period.quotaHours,
    adjustmentHours=adjustment, approvedHours=approved, pendingHours=pending, remainingHours=remaining,
  )
  return result


def getBalances(conn: Connection, actor: Actor, employeeId: int | None, day: date) -> list[dict]:
  targetId = employeeId if employeeId is not None else requireOwnEmployeeId(actor)
  employee = getEmployeeRow(conn, targetId)
  requireCanView(actor, employee)
  return [balanceFor(conn, employee["id"], employee["hired_on"], leaveType, day) for leaveType in listLeaveTypes(conn)]


# ---------- 額度調整 ----------


def createAdjustment(conn: Connection, actor: Actor, data: dict, now: datetime) -> dict:
  employee = getEmployeeRow(conn, data["employeeId"])
  leaveType = getLeaveType(conn, data["leaveTypeId"])
  if leaveType["quotaKind"] == "none":
    raise AppError("VALIDATION", "不限額度的假別不需要調整", 422, {"field": "leaveTypeId"})
  periodStart = data["periodStart"]
  period = rules.periodFor(leaveType["quotaKind"], leaveType["annualCapHours"], employee["hired_on"], periodStart)
  if period is None or period.start != periodStart:
    raise AppError("VALIDATION", "年度開始日不對：要填這位員工這個假別某個年度的第一天", 422, {"field": "periodStart"})
  hours = data["hours"]
  if not isinstance(hours, int) or hours == 0:
    raise AppError("VALIDATION", "時數不能是 0", 422, {"field": "hours"})
  reason = requireText(data.get("reason"), "原因")
  with mapDbErrors(conn):
    adjustmentId = conn.execute(
      "INSERT INTO leave_adjustments (employee_id, leave_type_id, period_start, hours, reason, created_by, created_at) "
      "VALUES (%s, %s, %s, %s, %s, %s, %s) RETURNING id",
      (employee["id"], leaveType["id"], periodStart, hours, reason, actor.userId, now),
    ).fetchone()["id"]
  adjustment = camelizeRow(conn.execute(f"{adjustmentSelect} WHERE a.id = %s", (adjustmentId,)).fetchone())
  writeAudit(conn, actor, "leave", "createLeaveAdjustment", "leaveAdjustment", adjustmentId, None, adjustment)
  return adjustment


def listAdjustments(conn: Connection, employeeId: int) -> list[dict]:
  rows = conn.execute(f"{adjustmentSelect} WHERE a.employee_id = %s ORDER BY a.period_start DESC, a.id DESC", (employeeId,)).fetchall()
  return camelizeRow(rows)


# ---------- 未休特休 ----------


def listUnused(conn: Connection, periodEndFrom: date, periodEndTo: date) -> list[dict]:
  if periodEndTo < periodEndFrom:
    raise AppError("VALIDATION", "結束日不能早於開始日", 422)
  row = conn.execute(f"SELECT {typeColumns} FROM leave_types WHERE quota_kind = 'seniority'").fetchone()
  if row is None:
    return []
  annual = camelizeRow(row)
  employees = conn.execute(
    "SELECT e.id, e.name, d.name AS department_name, e.hired_on, e.left_on "
    "FROM employees e JOIN departments d ON d.id = e.department_id ORDER BY e.employee_no"
  ).fetchall()
  result = []
  for employee in employees:
    day = rules.addMonths(employee["hired_on"], 6)
    while True:
      period = rules.seniorityPeriod(employee["hired_on"], day)
      # 離職後才開始的年度不算
      if period.end > periodEndTo or (employee["left_on"] and period.start > employee["left_on"]):
        break
      if period.end >= periodEndFrom:
        balance = balanceFor(conn, employee["id"], employee["hired_on"], annual, period.start)
        if balance["remainingHours"] > 0:
          result.append({
            "employeeId": employee["id"], "employeeName": employee["name"], "departmentName": employee["department_name"],
            "periodStart": balance["periodStart"], "periodEnd": balance["periodEnd"], "quotaHours": balance["quotaHours"],
            "adjustmentHours": balance["adjustmentHours"], "usedHours": balance["approvedHours"] + balance["pendingHours"],
            "unusedHours": balance["remainingHours"],
          })
      day = period.end + timedelta(days=1)
  return sorted(result, key=lambda item: (item["periodEnd"], item["employeeName"]))

def testConcurrentRequestsCannotTogetherExceedQuota(migratedUrl):

    try:

  for thread in threads:
    thread.join()
  assert sorted(outcomes) == ["LEAVE_QUOTA_EXCEEDED", "ok"], outcomes

from datetime import date, datetime

import psycopg
from psycopg import Connection

from app.infra.actor import Actor
from app.infra.db import savepoint
from app.infra.errors import AppError
from app.infra.serialize import camelizeRow
from app.modules.core.service import writeAudit
from app.modules.leave import rules, service

scopes = {"mine", "toApprove", "all"}
statuses = {"pending", "approved", "rejected", "withdrawn", "cancelled"}
requestSelect = (
  "SELECT r.id, r.employee_id, e.name AS employee_name, r.leave_type_id, t.name AS leave_type_name, r.start_at, r.end_at, "
  "r.hours, r.reason, r.status, r.decided_by, coalesce(de.name, du.email) AS decided_by_name, r.decided_at, r.decision_note, "
  "r.cancelled_by, r.cancelled_at, r.created_by, r.created_at "
  "FROM leave_requests r JOIN employees e ON e.id = r.employee_id JOIN leave_types t ON t.id = r.leave_type_id "
  "LEFT JOIN users du ON du.id = r.decided_by LEFT JOIN employees de ON de.id = du.employee_id"
)


@dataclass
class LeaveInput:
  leaveTypeId: int
  startAt: datetime
  endAt: datetime
  reason: str = ""
  acknowledgeWarnings: bool = False


def loadRequest(conn: Connection, requestId: int) -> dict:
  row = conn.execute(f"{requestSelect} WHERE r.id = %s", (requestId,)).fetchone()
  if row is None:
    raise AppError("NOT_FOUND", "找不到請假申請", 404)
  return camelizeRow(row)


def calendarKindsBetween(conn: Connection, startDay: date, endDay: date) -> dict[date, str]:
  rows = conn.execute("SELECT day, kind FROM calendar_days WHERE day BETWEEN %s AND %s", (startDay, endDay)).fetchall()
  return {row["day"]: row["kind"] for row in rows}


def findMeetingConflicts(conn: Connection, employeeId: int, startAt: datetime, endAt: datetime) -> list[dict]:
  rows = conn.execute(
    "SELECT b.id AS booking_id, b.title, b.start_at, b.end_at FROM booking_participants p JOIN bookings b ON b.id = p.booking_id "
    "WHERE p.employee_id = %s AND b.status = 'confirmed' AND b.start_at < %s AND b.end_at > %s ORDER BY b.start_at, b.id",
    (employeeId, endAt, startAt),
  ).fetchall()
  return camelizeRow(rows)


def evaluate(conn: Connection, actor: Actor, data: LeaveInput, lock: bool) -> dict:
  """試算與送出共用的檢查；回傳員工列、時數、送出後剩多少（不限額度時是 None）。"""
  employee = service.getEmployeeRow(conn, service.requireOwnEmployeeId(actor), lock=lock)
  if not employee["is_active"]:
    raise AppError("LEAVE_NOT_ELIGIBLE", "已離職的員工不能請假", 422)
  leaveType = service.getLeaveType(conn, data.leaveTypeId)
  if not leaveType["isActive"]:
    raise AppError("VALIDATION", "這個假別已經停用", 422, {"field": "leaveTypeId"})
  if data.endAt <= data.startAt:
    raise AppError("VALIDATION", "結束時間要晚於開始時間", 422, {"field": "endAt"})
  if data.startAt.date() < employee["hired_on"]:
    raise AppError("LEAVE_NOT_ELIGIBLE", "請假不能早於到職日", 422)
  kinds = calendarKindsBetween(conn, data.startAt.date(), data.endAt.date())
  hours = rules.countLeaveHours(data.startAt, data.endAt, kinds)
  if hours == 0:
    raise AppError("LEAVE_NO_WORKING_HOURS", "這段時間沒有上班時間，不需要請假", 422)
  period = rules.periodForRequest(leaveType["quotaKind"], leaveType["annualCapHours"], employee["hired_on"], data.startAt, data.endAt)
  remainingAfter = None
  if period.quotaHours is not None:
    balance = service.balanceFor(conn, employee["id"], employee["hired_on"], leaveType, period.start)
    remainingAfter = balance["remainingHours"] - hours
    if remainingAfter < 0:
      raise AppError(
        "LEAVE_QUOTA_EXCEEDED", f"{leaveType['name']}只剩 {balance['remainingHours']} 小時，這次要 {hours} 小時", 422,
        {"remainingHours": balance["remainingHours"]},
      )
  return {"employee": employee, "hours": hours, "remainingAfter": remainingAfter}


def quoteRequest(conn: Connection, actor: Actor, data: LeaveInput) -> dict:
  result = evaluate(conn, actor, data, lock=False)
  conflicts = findMeetingConflicts(conn, result["employee"]["id"], data.startAt, data.endAt)
  return {"hours": result["hours"], "remainingAfter": result["remainingAfter"], "conflicts": conflicts}


def createRequest(conn: Connection, actor: Actor, data: LeaveInput, now: datetime) -> dict:
  result = evaluate(conn, actor, data, lock=True)
  employeeId = result["employee"]["id"]
  if not data.acknowledgeWarnings:
    conflicts = findMeetingConflicts(conn, employeeId, data.startAt, data.endAt)
    if conflicts:
      raise AppError("MEETING_CONFLICT", "這段時間你有會議", 409, {"conflicts": conflicts})
  reason = data.reason.strip() if isinstance(data.reason, str) else ""
  try:
    with savepoint(conn):
      requestId = conn.execute(
        "INSERT INTO leave_requests (employee_id, leave_type_id, start_at, end_at, hours, reason, created_by, created_at) "
        "VALUES (%s, %s, %s, %s, %s, %s, %s, %s) RETURNING id",
        (employeeId, data.leaveTypeId, data.startAt, data.endAt, result["hours"], reason, actor.userId, now),
      ).fetchone()["id"]
  except psycopg.errors.ExclusionViolation:
    overlaps = conn.execute(
      f"{requestSelect} WHERE r.employee_id = %s AND r.status IN ('pending', 'approved') AND r.start_at < %s AND r.end_at > %s ORDER BY r.start_at",
      (employeeId, data.endAt, data.startAt),
    ).fetchall()
    raise AppError("LEAVE_OVERLAP", "這段時間已經有請假申請", 409, {"overlaps": camelizeRow(overlaps)})
  created = loadRequest(conn, requestId)
  writeAudit(conn, actor, "leave", "createLeaveRequest", "leaveRequest", requestId, None, created)
  return created


def listRequests(
  conn: Connection, actor: Actor, scope: str, status: str | None = None, start: datetime | None = None, end: datetime | None = None
) -> list[dict]:
  if scope not in scopes:
    raise AppError("VALIDATION", "scope 只能是 mine、toApprove、all", 422)
  if status is not None and status not in statuses:
    raise AppError("VALIDATION", f"status 只能是 {', '.join(sorted(statuses))}", 422)
  params = {"me": actor.employeeId, "isManager": service.isLeaveManager(actor), "status": status, "start": start, "end": end}
  joins = ""
  if scope == "mine":
    service.requireOwnEmployeeId(actor)
    condition = "r.employee_id = %(me)s"
  elif scope == "toApprove":
    joins = " LEFT JOIN employees m ON m.id = e.manager_id LEFT JOIN users mu ON mu.employee_id = m.id"
    # 主管離職、或主管沒有能登入的帳號時改由人資核准，否則這張申請沒有人看得到
    condition = (
      "r.status = 'pending' AND r.employee_id IS DISTINCT FROM %(me)s AND (e.manager_id = %(me)s "
      "OR (%(isManager)s AND (m.id IS NULL OR NOT m.is_active OR mu.id IS NULL OR NOT mu.is_active)))"
    )
  else:
    if not params["isManager"]:
      raise AppError("FORBIDDEN", "只有人資可以看所有人的請假", 403)
    condition = "true"
  rows = conn.execute(
    f"{requestSelect}{joins} WHERE {condition} AND (%(status)s::text IS NULL OR r.status = %(status)s) "
    "AND (%(start)s::timestamp IS NULL OR r.end_at > %(start)s) AND (%(end)s::timestamp IS NULL OR r.start_at < %(end)s) "
    "ORDER BY r.start_at DESC, r.id DESC",
    params,
  ).fetchall()
  return camelizeRow(rows)


def getRequest(conn: Connection, actor: Actor, requestId: int) -> dict:
  request = loadRequest(conn, requestId)
  service.requireCanView(actor, service.getEmployeeRow(conn, request["employeeId"]))
  return request

def lockForDecision(conn: Connection, requestId: int) -> tuple[dict, dict]:
  # 先鎖員工、再鎖申請，跟送出申請的順序一樣，兩邊同時進行時不會互相等待而死結
  employeeId = loadRequest(conn, requestId)["employeeId"]
  employee = service.getEmployeeRow(conn, employeeId, lock=True)
  conn.execute("SELECT id FROM leave_requests WHERE id = %s FOR UPDATE", (requestId,))
  return loadRequest(conn, requestId), employee


def requireStatus(request: dict, status: str, message: str) -> None:
  if request["status"] != status:
    raise AppError("LEAVE_INVALID_STATE", message, 409, {"status": request["status"]})


def requireApprover(conn: Connection, actor: Actor, request: dict) -> None:
  if actor.employeeId is not None and actor.employeeId == request["employeeId"]:
    raise AppError("FORBIDDEN", "不能核准自己的請假", 403)
  if service.isLeaveManager(actor):
    return
  employee = service.getEmployeeRow(conn, request["employeeId"])
  if actor.employeeId is not None and employee["manager_id"] == actor.employeeId:
    return
  raise AppError("FORBIDDEN", "只有直屬主管或人資可以核准這張申請", 403)


def approveRequest(conn: Connection, actor: Actor, requestId: int, note: str | None, now: datetime) -> dict:
  before, employee = lockForDecision(conn, requestId)
  requireApprover(conn, actor, before)
  requireStatus(before, "pending", "只有待核准的申請可以核准")
  leaveType = service.getLeaveType(conn, before["leaveTypeId"])
  startDay = datetime.fromisoformat(before["startAt"]).date()
  balance = service.balanceFor(conn, employee["id"], employee["hired_on"], leaveType, startDay)
  # 送出後人資可能調低了額度；這張本身已經算在待核准裡，剩餘小於 0 才是不夠
  if balance["remainingHours"] is not None and balance["remainingHours"] < 0:
    raise AppError(
      "LEAVE_QUOTA_EXCEEDED", f"{leaveType['name']}的額度已經不夠，差 {-balance['remainingHours']} 小時", 422,
      {"remainingHours": balance["remainingHours"]},
    )
  conn.execute(
    "UPDATE leave_requests SET status = 'approved', decided_by = %s, decided_at = %s, decision_note = %s WHERE id = %s",
    (actor.userId, now, (note or "").strip() or None, requestId),
  )
  after = loadRequest(conn, requestId)
  writeAudit(conn, actor, "leave", "approveLeaveRequest", "leaveRequest", requestId, before, after)
  return after


def rejectRequest(conn: Connection, actor: Actor, requestId: int, note: str | None, now: datetime) -> dict:
  before, _ = lockForDecision(conn, requestId)
  requireApprover(conn, actor, before)
  requireStatus(before, "pending", "只有待核准的申請可以退回")
  reason = requireText(note, "退回理由")
  conn.execute(
    "UPDATE leave_requests SET status = 'rejected', decided_by = %s, decided_at = %s, decision_note = %s WHERE id = %s",
    (actor.userId, now, reason, requestId),
  )
  after = loadRequest(conn, requestId)
  writeAudit(conn, actor, "leave", "rejectLeaveRequest", "leaveRequest", requestId, before, after)
  return after


def withdrawRequest(conn: Connection, actor: Actor, requestId: int, now: datetime) -> dict:
  before, _ = lockForDecision(conn, requestId)
  if actor.employeeId is None or actor.employeeId != before["employeeId"]:
    raise AppError("FORBIDDEN", "只有申請人可以撤回", 403)
  requireStatus(before, "pending", "只有待核准的申請可以撤回")
  conn.execute(
    "UPDATE leave_requests SET status = 'withdrawn', cancelled_by = %s, cancelled_at = %s WHERE id = %s", (actor.userId, now, requestId)
  )
  after = loadRequest(conn, requestId)
  writeAudit(conn, actor, "leave", "withdrawLeaveRequest", "leaveRequest", requestId, before, after)
  return after


def cancelRequest(conn: Connection, actor: Actor, requestId: int, now: datetime) -> dict:
  before, _ = lockForDecision(conn, requestId)
  isOwner = actor.employeeId is not None and actor.employeeId == before["employeeId"]
  isManager = service.isLeaveManager(actor)
  if not (isOwner or isManager):
    raise AppError("FORBIDDEN", "只有申請人或人資可以取消", 403)
  requireStatus(before, "approved", "只有已核准的請假可以取消")
  # 假已經開始或結束，可能牽涉薪資與出勤，交給人資處理
  if not isManager and datetime.fromisoformat(before["startAt"]) <= now:
    raise AppError("FORBIDDEN", "假已經開始，請找人資取消", 403)
  conn.execute(
    "UPDATE leave_requests SET status = 'cancelled', cancelled_by = %s, cancelled_at = %s WHERE id = %s", (actor.userId, now, requestId)
  )
  after = loadRequest(conn, requestId)
  writeAudit(conn, actor, "leave", "cancelLeaveRequest", "leaveRequest", requestId, before, after)
  return after

def listCalendar(conn: Connection, actor: Actor, start: datetime, end: datetime) -> list[dict]:
  if end <= start or end - start > timedelta(days=31):
    raise AppError("VALIDATION", "查詢區間要大於 0、最多 31 天", 422)
  rows = conn.execute(
    "SELECT r.id, r.employee_id, r.start_at, r.end_at, r.status, t.name AS leave_type_name, e.manager_id "
    "FROM leave_requests r JOIN employees e ON e.id = r.employee_id JOIN leave_types t ON t.id = r.leave_type_id "
    "WHERE r.status IN ('pending', 'approved') AND r.start_at < %s AND r.end_at > %s ORDER BY r.start_at, r.id",
    (end, start),
  ).fetchall()
  isManager = service.isLeaveManager(actor)
  result = []
  for row in rows:
    managerId = row.pop("manager_id")
    # 假別是私人資訊：只有本人、直屬主管、人資看得到，其他人只知道這段時間不在
    if not (isManager or (actor.employeeId is not None and actor.employeeId in (row["employee_id"], managerId))):
      row["leave_type_name"] = None
    result.append(row)
  return camelizeRow(result)


def findUnavailability(conn: Connection, employeeIds: list[int], startAt: datetime, endAt: datetime) -> list[dict]:
  rows = conn.execute(
    "SELECT employee_id, status, start_at, end_at FROM leave_requests WHERE employee_id = ANY(%s) "
    "AND status IN ('pending', 'approved') AND start_at < %s AND end_at > %s ORDER BY start_at, id",
    (employeeIds, endAt, startAt),
  ).fetchall()
  # 不帶假別：排會議的人不一定有權限知道別人請的是什麼假
  return camelizeRow([
    {"employeeId": row["employee_id"], "reason": "請假" if row["status"] == "approved" else "請假（待核准）",
     "startAt": row["start_at"], "endAt": row["end_at"]}
    for row in rows
  ])

function MeetingWarning({ error, pending, onOverride }: { error: ApiError | null; pending: boolean; onOverride(): void }) {
  if (error?.code !== "MEETING_CONFLICT") return null;

      </ul>

export function LeaveRequestForm({ onDone }: { onDone(): void }) {
  const toast = useToast();
  const { data: types } = useLeaveTypes();
  const [leaveTypeId, setLeaveTypeId] = useState("");
  const [startDate, setStartDate] = useState(tomorrow);
  const [startTime, setStartTime] = useState("09:00");
  const [endDate, setEndDate] = useState(tomorrow);
  const [endTime, setEndTime] = useState("18:00");
  const [reason, setReason] = useState("");
  const [quote, setQuote] = useState<LeaveQuote | null>(null);
  const [quoteError, setQuoteError] = useState<ApiError | null>(null);
  // 「仍要送出」只對下一次送出有效，不會悄悄套用到之後改過的內容
  const acknowledge = useRef(false);
  const startAt = `${startDate}T${startTime}`;
  const endAt = `${endDate}T${endTime}`;

  useEffect(() => {
    setQuote(null);
    setQuoteError(null);
    if (!leaveTypeId || !startDate || !endDate) return;
    let stale = false;
    const timer = setTimeout(() => {
      apiSend<LeaveQuote>("POST", "/api/leave/requests/quote", { leaveTypeId: Number(leaveTypeId), startAt, endAt })
        .then((result) => { if (!stale) setQuote(result); })
        .catch((caught) => { if (!stale) setQuoteError(caught instanceof ApiError ? caught : new ApiError(0, "UNKNOWN", "試算失敗")); });
    }, quoteDelay);
    return () => { stale = true; clearTimeout(timer); };
  }, [leaveTypeId, startAt, endAt, startDate, endDate]);

  const { submit, pending, error, fieldErrors } = useSubmit(
    () => {
      const ack = acknowledge.current;
      acknowledge.current = false;
      return apiSend("POST", "/api/leave/requests", { leaveTypeId: Number(leaveTypeId), startAt, endAt, reason, acknowledgeWarnings: ack });
    },
    { onSuccess: () => { toast("已送出請假申請"); onDone(); }, invalidate: [["leave"]] },
  );

  function onSubmit(event: FormEvent) {
    event.preventDefault();
    void submit();
  }

  const typeOptions = (types ?? []).map((item) => ({ value: String(item.id), label: item.name }));
  const quoteText = quote && `共 ${quote.hours} 小時${quote.remainingAfter === null ? "" : `，送出後剩 ${quote.remainingAfter} 小時`}`;

  return (
    <form className="form" onSubmit={onSubmit}>
      <SelectField label="假別" value={leaveTypeId} onChange={setLeaveTypeId} options={typeOptions} allowEmpty="請選擇" required error={fieldErrors.leaveTypeId} />
      <TextField label="開始日期" type="date" required value={startDate} onChange={setStartDate} />
      <SelectField label="開始時間" value={startTime} onChange={setStartTime} options={hourOptions} error={fieldErrors.startAt} />
      <TextField label="結束日期" type="date" required value={endDate} onChange={setEndDate} />
      <SelectField label="結束時間" value={endTime} onChange={setEndTime} options={hourOptions} error={fieldErrors.endAt} />
      <TextArea label="事由" value={reason} onChange={setReason} />
      {quoteText && <div className="formNote">{quoteText}</div>}
      {quote && quote.conflicts.length > 0 && (
        <div className="formWarn">這段時間你有會議：{quote.conflicts.map((item) => item.title).join("、")}</div>
      )}
      <FormError error={quoteError} />
      <MeetingWarning error={error} pending={pending} onOverride={() => { acknowledge.current = true; void submit(); }} />
      {error?.code !== "MEETING_CONFLICT" && <FormError error={error} />}
      <div className="formActions">
        <button className="primary" type="submit" disabled={pending || quoteError !== null || !leaveTypeId}>送出</button>
        <button className="secondary" type="button" onClick={onDone}>取消</button>
      </div>
    </form>
  );
}

// 二次確認：第一下變成「確定…？」，第二下才送出；連點的第二下（detail > 1）不算確認
function ConfirmButton({ label, confirmLabel, pending, onConfirm }: { label: string; confirmLabel: string; pending: boolean; onConfirm(): void }) {
  const [confirming, setConfirming] = useState(false);
  return (
    <button className="danger" type="button" disabled={pending} onClick={(event) => {
      if (event.detail > 1) return;
      if (confirming) onConfirm(); else setConfirming(true);
    }}>
      {confirming ? confirmLabel : label}
    </button>
  );
}

export function OwnerActions({ request, onDone }: { request: LeaveRequest; onDone(): void }) {
  const toast = useToast();
  const { user } = useAuth();
  const { can } = usePermissions();
  const withdraw = useSubmit(
    () => apiSend("POST", `/api/leave/requests/${request.id}/withdraw`),
    { onSuccess: () => { toast("已撤回申請"); onDone(); }, invalidate: [["leave"]] },
  );
  const cancel = useSubmit(
    () => apiSend("POST", `/api/leave/requests/${request.id}/cancel`),
    { onSuccess: () => { toast("已取消請假"); onDone(); }, invalidate: [["leave"]] },
  );
  const isOwner = user?.employeeId === request.employeeId;
  const notStarted = parseLocal(request.startAt) > new Date();
  const canWithdraw = isOwner && request.status === "pending";
  // 假開始後只有人資能取消：可能已經牽涉出勤與薪資
  const canCancel = request.status === "approved" && ((isOwner && notStarted) || can("leave.manage"));
  return (
    <>
      <FormError error={withdraw.error ?? cancel.error} />
      {canWithdraw && <ConfirmButton label="撤回" confirmLabel="確定撤回？" pending={withdraw.pending} onConfirm={() => void withdraw.submit()} />}
      {canCancel && <ConfirmButton label="取消請假" confirmLabel="確定取消？" pending={cancel.pending} onConfirm={() => void cancel.submit()} />}
    </>
  );
}

export function LeaveTypeForm({ leaveType, onDone }: { leaveType?: LeaveType; onDone(): void }) {
  const toast = useToast();
  const [code, setCode] = useState(leaveType?.code ?? "");
  const [name, setName] = useState(leaveType?.name ?? "");
  const [paidRatio, setPaidRatio] = useState(String(leaveType?.paidRatio ?? 1));
  const [quotaKind, setQuotaKind] = useState<QuotaKind>(leaveType?.quotaKind ?? "yearly_cap");
  const [capHours, setCapHours] = useState(leaveType?.annualCapHours ? String(leaveType.annualCapHours) : "");
  const { submit, pending, error, fieldErrors } = useSubmit(
    () => {
      const annualCapHours = quotaKind === "yearly_cap" ? Number(capHours) : null;
      if (!leaveType) return apiSend("POST", "/api/leave/types", { code, name, paidRatio: Number(paidRatio), quotaKind, annualCapHours });
      // 額度類型與代碼建立後不能改，只送可以改的欄位
      const changed: Record<string, unknown> = {};
      if (name !== leaveType.name) changed.name = name;
      if (Number(paidRatio) !== leaveType.paidRatio) changed.paidRatio = Number(paidRatio);
      if (quotaKind === "yearly_cap" && annualCapHours !== leaveType.annualCapHours) changed.annualCapHours = annualCapHours;
      if (Object.keys(changed).length === 0) return Promise.resolve(undefined);
      return apiSend("PATCH", `/api/leave/types/${leaveType.id}`, changed);
    },
    { onSuccess: () => { toast(leaveType ? "已更新假別" : "已新增假別"); onDone(); }, invalidate: [["leave"]] },
  );
  return (
    <form className="form" onSubmit={(event: FormEvent) => { event.preventDefault(); void submit(); }}>

const workStart = "09:00:00";
const workEnd = "18:00:00";

// 跨好幾天的假在中間那幾天會蓋滿 00:00–24:00，把時間軸撐成整天；裁到當天的上班時間才看得清楚
export function clipToWorkday(item: { startAt: string; endAt: string }, day: string): { startAt: string; endAt: string } | null {
  const windowStart = `${day}T${workStart}`;
  const windowEnd = `${day}T${workEnd}`;
  // 同格式的本地時間字串可以直接比大小
  const startAt = item.startAt > windowStart ? item.startAt : windowStart;
  const endAt = item.endAt < windowEnd ? item.endAt : windowEnd;
  return startAt < endAt ? { startAt, endAt } : null;
}

export function toLeaveTimelineItems(items: LeaveCalendarItem[], from: string): TimelineItem[] {
  const day = from.slice(0, 10);
  return items.flatMap((item) => {
    const clipped = clipToWorkday(item, day);
    if (!clipped) return [];
    const pending = item.status === "pending";
    return [{
      key: `leave:${item.id}`,
      kind: "block" as const,
      subjectId: item.employeeId,
      ...clipped,
      label: `${item.leaveTypeName ?? "請假"}${pending ? "（待核准）" : ""}`,
      muted: pending,
      // 看不到假別的人也看不到詳情，不給連結
      link: item.leaveTypeName === null ? undefined : `/leave?id=${item.id}`,
    }];
  });
}
